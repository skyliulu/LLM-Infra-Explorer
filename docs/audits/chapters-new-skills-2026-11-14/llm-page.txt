def prefill(request_ids, input_tokens, kv_cache, temp=0.7, top_k=3, top_p=0.9):
  # Embedding
  meta = scheduler.inference_meta(request_ids)
  x = embedding(input_tokens) # [L, d]


for layer_id in range(32): # 层循环堆叠
      # 注意力与 KV Cache
      layer_cache = kv_cache.layer(layer_id)
      q, k, v = qkv_proj[layer_id](norm(x))
      q, k = apply_rope(q, k, meta.positions) # RoPE 在每层 Attention 内旋转 Q/K
      slots = layer_cache.reserve(meta.slot_mapping)
      layer_cache.write_(slots, k, v)
      backend = select_attention_backend(meta, q.dtype)
      attn_out = backend.prefill(q, layer_cache, meta)
      x = x + o_proj[layer_id](attn_out)


      # Dense FFN
      ffn_in = norm(x)
      hidden = gelu(ffn_in @ W_up[layer_id])
      x = x + (hidden @ W_down) # 残差连接并进入下一层


  # LM Head & 温度采样
  logits = x[-1] @ W_vocab # 映射到词表
  logits = logits / temp   # 温度调整缩放
  probs = softmax(logits)
  probs = top_k_filter(probs, top_k)
  probs = top_p_filter(probs, top_p)
  return sample(renormalize(probs))
