def forward_layer(x, request_meta):


  # DP模式: 仅接收属于当前 Rank 的局部子批次 (B/4)
  x_owned = select_owned_requests(x, request_meta)


  # Q 来自隐藏状态；不能从 KV latent 生成
  q = q_proj(x_owned)
  c_kv, k_rope = kv_proj(x_owned)
  # 每个 Token 持久化压缩 KV latent 与 RoPE Key
  kv_cache.write(request_meta.slots, c_kv, k_rope)
  attn_local = mla_decode(q, kv_cache.local_view(request_meta))


  # 非 EP 路径：DP 请求分片聚合为 TP-FFN 输入
  x_global = dp_gather(attn_local, request_meta.dp_sizes)


  # TP-FFN：每个 Rank 持有参数分片
  expert_out = local_experts(x_global)


  # 归约 TP 局部结果，并直接返回请求所属 DP Rank
  x_local = dp_reduce_scatter(expert_out)


  # 提交本地请求的层输出
  return x_local
