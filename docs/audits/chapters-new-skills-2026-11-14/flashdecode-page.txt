底层代码 (Python 伪代码)
O + LSE 工作区
def flash_decoding_lse(q, k, v, num_splits):
  kv_head = q_head // q_per_kv_head # Q head 映射到共享 KV head
  k, v = resolve_kv(k, v, block_table) # 连续 KV：基址 + token 偏移
# [步骤 1] 建立逻辑 Split 并分配 CTA 工作
  partials = []
  for i in range(num_splits):
# [步骤 2] Kernel 1: 分块局部计算 LSE
      scores = (q @ k[i].T) / sqrt(d)
      m_i = max(scores)
      l_i = sum(exp(scores - m_i))
      O_i = (exp(scores - m_i) @ v[i]) / l_i
      L_i = m_i + log(l_i) # 关键：Log-Sum-Exp
      partials.append((O_i, L_i)) # 写入 HBM Workspace
# [步骤 3] Kernel 2 (归约): 迭代求全局 LSE
  L_global = partials[0].L
  for i in range(1, num_splits):
      L_max = max(L_global, partials[i].L)
      L_min = min(L_global, partials[i].L)
      L_global = L_max + log1p(exp(L_min - L_max))
# [步骤 4] 利用 L_global 合并各 Split
  O_global = 0
  for O_i, L_i in partials:
      weight = exp(L_i - L_global) # 权重对齐
      O_global += O_i * weight
  return O_global # 写回结果
